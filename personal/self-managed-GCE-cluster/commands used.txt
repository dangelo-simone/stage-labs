# --- prep script ---

cat > node-prep.sh <<'EOF'
#!/usr/bin/env bash
set -euo pipefail

echo ">>> [1/5] Disabling swap"
sudo swapoff -a
sudo sed -i '/\bswap\b/ s/^\([^#]\)/#\1/' /etc/fstab   # usually a no-op on GCE

echo ">>> [2/5] Loading kernel modules (overlay, br_netfilter)"
cat <<MOD | sudo tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
MOD
sudo modprobe overlay
sudo modprobe br_netfilter

echo ">>> [3/5] Networking sysctls"
cat <<SYS | sudo tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
SYS
sudo sysctl --system >/dev/null

echo ">>> [4/5] Installing and configuring containerd"
sudo apt-get update
sudo apt-get install -y containerd
sudo mkdir -p /etc/containerd
containerd config default | sudo tee /etc/containerd/config.toml >/dev/null
# kubelet and containerd must agree on the cgroup driver -> systemd
sudo sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml
sudo systemctl restart containerd
sudo systemctl enable containerd

echo ">>> [5/5] Installing kubelet, kubeadm, kubectl (v1.31)"
sudo apt-get install -y apt-transport-https ca-certificates curl gpg
sudo mkdir -p /etc/apt/keyrings
curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.31/deb/Release.key \
  | sudo gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
echo 'deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.31/deb/ /' \
  | sudo tee /etc/apt/sources.list.d/kubernetes.list
sudo apt-get update
sudo apt-get install -y kubelet kubeadm kubectl
sudo apt-mark hold kubelet kubeadm kubectl   # pin versions so an apt upgrade can't break the cluster
sudo systemctl enable --now kubelet

echo ">>> Node prep complete."
EOF

#spiegazione: kube-proxy is a component running on every node that implements Kubernetes Services. We need it because a Service is just a stable virtual IP/port (a ClusterIP, or a NodePort on every node) — nothing actually listens there; kube-proxy is what makes that address route to real pods. It works by watching the API server for Services and their endpoints (the healthy pods) and programming the node's kernel packet-filtering rules (iptables by default) to DNAT any connection arriving at the Service address to one of those pods, picked at random per connection. So it's a node-level, Layer-4 load balancer over a Service's pods: it doesn't proxy traffic in userspace, it just installs kernel rules that rewrite the destination — which is why hitting a NodePort spreads connections across pods (and why the pick can land on a pod on another node).

#architettura
Internet :80
   │
   ▼
GCP external passthrough NLB  (L4: forwarding rule :80 → workers :80)
   │
   ▼
ingress-nginx on each worker's host :80   (L7: reads Ingress objects, routes by host/path)
   │
   ▼

echo "Public IP: $INGRESS_IP"

# 2. Group the workers so the LB has something to point at.
gcloud compute instance-groups unmanaged create k8s-workers-ig --zone="$ZONE"
gcloud compute instance-groups unmanaged add-instances k8s-workers-ig \
  --zone="$ZONE" --instances=k8s-worker-1,k8s-worker-2

# 3. Health check: hit ingress-nginx's own health endpoint (:10254/healthz).
gcloud compute health-checks create http k8s-ingress-hc \
  --region="$REGION" --port=10254 --request-path=/healthz

# 4. Regional external backend service (L4 passthrough, TCP), wired to the HC.
gcloud compute backend-services create k8s-ingress-bes \
  --load-balancing-scheme=EXTERNAL --protocol=TCP --region="$REGION" \
  --health-checks=k8s-ingress-hc --health-checks-region="$REGION"
gcloud compute backend-services add-backend k8s-ingress-bes \
  --region="$REGION" --instance-group=k8s-workers-ig --instance-group-zone="$ZONE"

# 5. Forwarding rule: public IP :80 -> backend. Passthrough preserves the port,
#    so it lands on the workers' :80, exactly where nginx is listening.
gcloud compute forwarding-rules create k8s-ingress-fr \
  --region="$REGION" --load-balancing-scheme=EXTERNAL \
  --address=k8s-ingress-ip --ip-protocol=TCP --ports=80 \
  --backend-service=k8s-ingress-bes

# StorageClass with NO provisioner — we're binding pre-made volumes, not creating them.
# WaitForFirstConsumer delays binding until a pod is scheduled, so the scheduler
# respects the PV's node affinity.
cat > ~/k8s-storage/storageclass.yaml <<'EOF'
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: local-storage
provisioner: kubernetes.io/no-provisioner
volumeBindingMode: WaitForFirstConsumer
EOF

# The PersistentVolume: "there is 10Gi at /mnt/disks/k8s-data, but ONLY on worker-1."
cat > ~/k8s-storage/pv.yaml <<'EOF'
apiVersion: v1
kind: PersistentVolume
metadata:
  name: pg-data-pv
spec:
  capacity:
    storage: 10Gi
  accessModes: [ReadWriteOnce]
  persistentVolumeReclaimPolicy: Retain
  storageClassName: local-storage
  local:
    path: /mnt/disks/k8s-data
  nodeAffinity:
    required:
      nodeSelectorTerms:
        - matchExpressions:
            - key: kubernetes.io/hostname
              operator: In
              values: [k8s-worker-1]
EOF

              mountPath: /var/lib/postgresql/data
      volumes:
        - name: data
          persistentVolumeClaim:
            claimName: pg-data
EOF

kubectl apply -f ~/k8s-storage/storageclass.yaml
kubectl apply -f ~/k8s-storage/pv.yaml
kubectl apply -f ~/k8s-storage/postgres.yaml
